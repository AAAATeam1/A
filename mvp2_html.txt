<script>
  // الشاشات
  const registerScreen = document.getElementById("registerScreen");
  const monitorScreen  = document.getElementById("monitorScreen");

  // التسجيل
  const regName = document.getElementById("regName");
  const regId   = document.getElementById("regId");
  const regAge  = document.getElementById("regAge");
  const startBtn = document.getElementById("startBtn");

  // بيانات المريض
  const pName = document.getElementById("pName");
  const pId   = document.getElementById("pId");
  const pAge  = document.getElementById("pAge");

  // القراءة
  const readingValue = document.getElementById("readingValue");
  const glucoseInput = document.getElementById("glucoseInput");
  const updateBtn    = document.getElementById("updateBtn");

  // الشريط العلوي
  const greenHeader   = document.getElementById("greenHeader");
  const bigValue      = document.getElementById("bigValue");
  const bigStatusText = document.getElementById("bigStatusText");

  // التنبيه
  const statusPill  = document.getElementById("statusPill");
  const alertBox    = document.getElementById("alertBox");
  const alertIcon   = document.getElementById("alertIcon");
  const alertTitle  = document.getElementById("alertTitle");
  const alertBody   = document.getElementById("alertBody");
  const alertActions = document.getElementById("alertActions");
  const alertNote   = document.getElementById("alertNote");
  const timerCircle = document.getElementById("timerCircle");
  const okBtn       = document.getElementById("okBtn");
  const helpBox     = document.getElementById("helpBox");

  let countdown = null;
  let timeLeft  = 60;

  function resetAlert() {
    if (countdown) {
      clearInterval(countdown);
      countdown = null;
    }
    timeLeft = 60;
    timerCircle.textContent = "60";
    alertActions.style.display = "none";
    alertNote.style.display    = "none";
    helpBox.style.display      = "none";
    helpBox.style.background   = "transparent";
    helpBox.style.color        = "#9ca3af";
    helpBox.style.border       = "1px dashed rgba(148,163,184,0.8)";
    helpBox.textContent        = "";
  }

  // عدّاد التصعيد: mode = "low" أو "high"
  function startCountdown(mode) {
    alertActions.style.display = "flex";
    alertNote.style.display    = "block";
    timeLeft = 60;
    timerCircle.textContent = "60";

    countdown = setInterval(() => {
      timeLeft--;
      if (timeLeft < 0) timeLeft = 0;
      timerCircle.textContent = timeLeft;

      // عند انتهاء العدّاد
      if (timeLeft <= 0) {
        clearInterval(countdown);
        countdown = null;

        helpBox.style.display    = "block";
        helpBox.style.background = "#facc15";   // أصفر
        helpBox.style.color      = "#1f2937";
        helpBox.style.border     = "1px solid #eab308";

        if (mode === "low") {
          helpBox.textContent =
            "يحتاج إلى جلوكوز فوراً + تم مشاركة الموقع مع الجهات المختصة.";
        } else {
          helpBox.textContent =
            "يحتاج إلى أنسولين + تم مشاركة الموقع مع الجهات المختصة.";
        }

        alertTitle.textContent =
          mode === "low"
            ? "انخفاض شديد – تم التصعيد"
            : "ارتفاع شديد – تم التصعيد";

        alertBody.textContent =
          "لم يتم التفاعل خلال المهلة المحددة. تم إرسال البلاغ افتراضياً للجهات المختصة مع مشاركة الموقع الجغرافي.";
        alertNote.style.display = "none";
      }
    }, 1000);
  }

  // بدء المتابعة
  startBtn.onclick = () => {
    const name = regName.value.trim();
    if (!name) {
      alert("الرجاء إدخال اسم المريض.");
      return;
    }

    pName.textContent = "المريض: " + name;
    pId.textContent   = "الهوية: " + (regId.value.trim() || "غير مسجل");
    pAge.textContent  = "العمر: " + (regAge.value.trim() || "—");

    registerScreen.style.display = "none";
    monitorScreen.style.display  = "block";
  };

  /* ----------------- الرسم البياني ----------------- */

  const chartCtx = document.getElementById("glucoseChart").getContext("2d");
  let labels = [];
  let dataVals = [];

  const greenBand = {
    id: "greenBand",
    beforeDraw(chart) {
      const {ctx, chartArea, scales:{y}} = chart;
      const top    = y.getPixelForValue(150);
      const bottom = y.getPixelForValue(76);
      ctx.save();
      ctx.fillStyle = "rgba(74,222,128,0.18)";
      ctx.fillRect(chartArea.left, top, chartArea.right - chartArea.left, bottom - top);
      ctx.restore();
    }
  };

  const glucoseChart = new Chart(chartCtx, {
    type: "line",
    data: {
      labels: labels,
      datasets: [{
        data: dataVals,
        borderColor: "#111827",
        borderWidth: 2,
        tension: 0.35,
        pointRadius: 4,
        pointBackgroundColor: "#22c55e"
      }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      scales: {
        y: {
          min: 0,
          max: 250,
          ticks: { font: { size: 10 } },
          grid: { color: "#e5e7eb" }
        },
        x: {
          ticks: { font: { size: 9 } },
          grid: { display: false }
        }
      },
      plugins: {
        legend: { display: false }
      }
    },
    plugins: [greenBand]
  });

  function addReadingToChart(value) {
    const label = new Date().toLocaleTimeString("ar-SA", {
      hour: "2-digit",
      minute: "2-digit"
    });
    labels.push(label);
    dataVals.push(value);
    if (labels.length > 8) {
      labels.shift();
      dataVals.shift();
    }
    glucoseChart.update();
  }

  /* ----------------- منطق النطاقات ----------------- */

  updateBtn.onclick = () => {
    const value = parseFloat(glucoseInput.value);
    if (isNaN(value)) {
      alert("الرجاء إدخال قيمة رقمية صحيحة.");
      return;
    }

    readingValue.textContent = value.toString();
    bigValue.textContent     = value.toString();
    addReadingToChart(value);

    resetAlert();
    alertBox.style.display = "none";
    statusPill.style.display = "inline-flex";
    statusPill.className = "pill";
    statusPill.innerHTML = "";

    // ≤56       → أحمر خطير منخفض + عدّاد
    // 57–60     → أحمر منخفض (رسالة فقط)
    // 61–75     → برتقالي آمن (بدون تنبيه)
    // 76–150    → أخضر معتدل (بدون تنبيه)
    // 151–198   → برتقالي آمن (بدون تنبيه)
    // 199–350   → أحمر مرتفع (بدون عدّاد)
    // >350      → أحمر مرتفع جدًا مع عدّاد

    if (value <= 56) {
      statusPill.classList.add("pill-red");
      statusPill.innerHTML = '<span class="icon">🚨</span><span>انخفاض شديد</span>';

      greenHeader.style.background = "#b91c1c";
      greenHeader.style.color      = "#fee2e2";
      bigStatusText.textContent    = "انخفاض شديد في السكر – يحتاج جلوكوز فورًا.";

      alertBox.style.display = "block";
      alertBox.className = "alert-box alert-red";
      alertIcon.textContent = "🚨";
      alertTitle.textContent = "الجلوكوز منخفض جدًا وخطير";
      alertBody.textContent =
        "السكر أقل من أو يساوي ٥٦ mg/dL – يحتاج إلى جلوكوز فوراً ومتابعة . سيتم التصعيد خلال ٦٠ ثانية إذا لم يتم التفاعل.";
      startCountdown("low");
    }
    else if (value >= 57 && value <= 60) {
      statusPill.classList.add("pill-red");
      statusPill.innerHTML = '<span class="icon">⚠️</span><span>منخفض</span>';

      greenHeader.style.background = "#f97316";
      greenHeader.style.color      = "#fff7ed";
      bigStatusText.textContent    = "الجلوكوز منخفض – يُنصح بتناول جلوكوز وإعادة القياس.";

      alertBox.style.display = "block";
      alertBox.className = "alert-box alert-red";
      alertIcon.textContent = "⚠️";
      alertTitle.textContent = "الجلوكوز منخفض";
      alertBody.textContent =
        "القراءة بين ٥٧ و٦٠ mg/dL – يُنصح بتناول مصدر جلوكوز وإعادة القياس خلال دقائق.";
    }
    else if (value >= 61 && value <= 75) {
      statusPill.classList.add("pill-orange");
      statusPill.innerHTML = '<span class="icon">🟠</span><span>نطاق برتقالي آمن</span>';

      greenHeader.style.background = "#f97316";
      greenHeader.style.color      = "#fff7ed";
      bigStatusText.textContent    = "نطاق برتقالي آمن – المتابعة مستمرة بدون تنبيه.";
    }
    else if (value >= 76 && value <= 150) {
      statusPill.classList.add("pill-green");
      statusPill.innerHTML = '<span class="icon">🟢</span><span>نطاق أخضر – السكر معتدل</span>';

      greenHeader.style.background = "#4ade80";
      greenHeader.style.color      = "#064e3b";
      bigStatusText.textContent    = "السكر ضمن النطاق المعتدل.";
    }
    else if (value >= 151 && value <= 198) {
      statusPill.classList.add("pill-orange");
      statusPill.innerHTML = '<span class="icon">🟠</span><span>نطاق برتقالي آمن – يحتاج متابعة</span>';

      greenHeader.style.background = "#f97316";
      greenHeader.style.color      = "#fff7ed";
      bigStatusText.textContent    = "ارتفاع بسيط – يحتاج متابعة حسب خطة الطبيب.";
    }
    else if (value >= 199 && value <= 350) {
      statusPill.classList.add("pill-red");
      statusPill.innerHTML = '<span class="icon">🚨</span><span>مرتفع</span>';

      greenHeader.style.background = "#b91c1c";
      greenHeader.style.color      = "#fee2e2";
      bigStatusText.textContent    = "الجلوكوز مرتفع – يُنصح بمراجعة الخطة العلاجية.";

      alertBox.style.display = "block";
      alertBox.className = "alert-box alert-red";
      alertIcon.textContent = "🚨";
      alertTitle.textContent = "الجلوكوز مرتفع";
      alertBody.textContent =
        "القراءة في النطاق الأحمر المرتفع (حوالي ١٩٩ – ٣٥٠ mg/dL). يُنصح بمراجعة الخطة العلاجية والتواصل مع الطبيب إذا تكررت هذه القراءات.";
    }
    else if (value > 350) {
      statusPill.classList.add("pill-red");
      statusPill.innerHTML = '<span class="icon">🚨</span><span>مرتفع جدًا وخطر</span>';

      greenHeader.style.background = "#7f1d1d";
      greenHeader.style.color      = "#fee2e2";
      bigStatusText.textContent    = "الجلوكوز مرتفع جدًا – خطر حموضة الدم ويحتاج تدخل طبي عاجل.";

      alertBox.style.display = "block";
      alertBox.className = "alert-box alert-red";
      alertIcon.textContent = "🚨";
